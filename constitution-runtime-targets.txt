0542:       const data = noteResult.evidence[0].attributes;
0543:       return `Created and verified ${data.filename}. The SHA-256 evidence matches the real file, and this receipt can undo the action while the file remains unchanged.`;
0544:     }
0545:     const liveResult = task.results.find((result) => result.capability_id === "workspace.read_status");
0546:     if (liveResult?.evidence?.[0]) {
0547:       const data = liveResult.evidence[0].attributes;
0548:       const cleanText = data.git_clean ? "clean" : `${data.git_changed_entries} changed entries`;
0549:       return `Verified live workspace status. Branch ${data.git_branch} at ${String(data.git_commit).slice(0, 8)}; working tree ${cleanText}; ${Math.round(Number(data.disk_free_bytes) / 1073741824)} GB disk space free.`;
0550:     }
0551:     return `Completed and verified ${task.results.length} capability result${task.results.length === 1 ? "" : "s"}.`;
0552:   }
0553:   if (task.state === "awaiting_approval") {
0554:     if (task.approval?.approval_channel === "phone") {
0555:       return "I prepared a signed trusted-device command. No command has executed. Pair your phone and approve the exact payload there.";
0556:     }
0557:     return "I prepared a controlled write action. No file has been created. Review the exact payload and approve or cancel it.";
0558:   }
0559:   if (task.state === "rolled_back") {
0560:     const deviceRollback = task.results.some((result) => result.capability_id === "device.rollback_launch_notepad");
0561:     return deviceRollback
0562:       ? "Undo completed. The receipt-bound device process was terminated and verified absent."
0563:       : "Undo completed. The receipt-owned note was removed and its absence was verified.";
0564:   }
0565:   if (task.state === "denied") {
0566:     const reason = task.policy_decisions.find((decision) => decision.outcome === "deny")?.reason_code;
0567:     return `The request was denied by policy${reason ? `: ${reason}` : ""}.`;
0568:   }
0569:   if (task.state === "failed") {
0570:     /*
0571:      * Name the reason. "Review the evidence panel" makes the person do the
0572:      * diagnosis the system already did, and hides whether the cause was a
0573:      * provider error, a validation refusal, or a missing dependency.
0574:      */
0575:     const failure = task.results.find((result) => result.error_code);
0576:     if (failure) {
0577:       return {
0578:         text: `I stopped without acting. ${failureExplanation(String(failure.error_code))}`,
0579:         blocks: [{ type: "note", text: `Capability ${failure.capability_id} reported ${failure.error_code}.` }],
0580:       };
0581:     }
0582:     return "The action failed closed, and no capability recorded a reason. The lifecycle panel has the event trail.";
0583:   }
0584:   return `The task is currently ${titleCase(task.state)}.`;
0585: }

0607: async function pollTaskUntilResolved(taskId) {
0608:   if (taskPollTimer) clearInterval(taskPollTimer);
0609:   taskPollTimer = setInterval(async () => {
0610:     try {
0611:       const response = await fetch(`/v1/tasks/${taskId}`, { cache: "no-store" });
0612:       if (!response.ok) return;
0613:       const task = await response.json();
0614:       if (task.state === "awaiting_approval") return;
0615:       clearInterval(taskPollTimer);
0616:       taskPollTimer = null;
0617:       const receipt = await fetchReceipt(task.task_id);
0618:       hideApproval();
0619:       renderTask(task, receipt);
0620:       addMessage("assistant", summarizeTask(task));
0621:       showToast(task.state === "completed" ? "Phone approval accepted. Device command verified." : `Phone decision: ${titleCase(task.state)}.`);
0622:     } catch (_error) {
0623:       // Health polling and the phone client remain the source of truth during transient errors.
0624:     }
0625:   }, 1200);
0626: }
